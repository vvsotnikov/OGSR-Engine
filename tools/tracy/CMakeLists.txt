cmake_minimum_required(VERSION 3.24)
project(ogsr-trace-tools LANGUAGES C CXX)
if(NOT TRACY_SOURCE)
    message(FATAL_ERROR "Set TRACY_SOURCE to the pinned Tracy checkout")
endif()
set(CMAKE_CXX_STANDARD 20)
set(NO_STATISTICS OFF)
option(NO_ISA_EXTENSIONS "Disable native instruction extensions" OFF)
include("${TRACY_SOURCE}/cmake/config.cmake")
include("${TRACY_SOURCE}/cmake/vendor.cmake")
include("${TRACY_SOURCE}/cmake/server.cmake")
add_executable(ogsr-trace-export trace_export.cpp)
target_link_libraries(ogsr-trace-export PRIVATE TracyServer)
