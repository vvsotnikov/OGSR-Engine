cmake_minimum_required(VERSION 3.16)
project(scheduler-regression LANGUAGES CXX)
# Deterministic clocks and mock game objects validate scheduling decisions.
# C++ catch substitution does not test Windows SEH behavior; build the engine too.
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
    "${ROOT}/ogsr_engine/xr_3da/xrSheduler.h" "${ROOT}/ogsr_engine/xr_3da/xrSheduler.cpp"
    "${CMAKE_CURRENT_SOURCE_DIR}/Baseline.inc")
file(READ "${ROOT}/ogsr_engine/xr_3da/xrSheduler.h" HEADER)
string(REPLACE "#include \"ISheduled.h\"" "" HEADER "${HEADER}")
file(READ "${ROOT}/ogsr_engine/xr_3da/xrSheduler.cpp" SOURCE)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/Baseline.inc" BASELINE)
# Exact unique function boundaries, not brace counting or a historical Git lookup.
foreach(MARKER "void CSheduler::ProcessStep()" "void CSheduler::Update()")
    string(FIND "${SOURCE}" "${MARKER}" FIRST)
    string(FIND "${SOURCE}" "${MARKER}" LAST REVERSE)
    if(FIRST LESS 0 OR NOT FIRST EQUAL LAST)
        message(FATAL_ERROR "Missing or ambiguous scheduler boundary: ${MARKER}")
    endif()
endforeach()
string(FIND "${SOURCE}" "void CSheduler::ProcessStep()" BEGIN)
string(FIND "${SOURCE}" "void CSheduler::Update()" END)
if(NOT BEGIN LESS END)
    message(FATAL_ERROR "Unexpected scheduler function ordering")
endif()
string(SUBSTRING "${SOURCE}" 0 ${BEGIN} PREFIX)
string(SUBSTRING "${SOURCE}" ${END} -1 SUFFIX)
set(LEGACY "${PREFIX}${BASELINE}${SUFFIX}")
foreach(VARIANT current legacy)
    if(VARIANT STREQUAL "current")
        set(CODE "${SOURCE}")
    else()
        set(CODE "${LEGACY}")
    endif()
    string(REGEX REPLACE "#include[^\n]*" "" CODE "${CODE}")
    string(REPLACE "__try" "try" CODE "${CODE}")
    string(REPLACE "__except (ExceptStackTrace(\"[CSheduler::ProcessStep] stack trace:\\n\"))" "catch (...)" CODE "${CODE}")
    string(REPLACE "__except (ExceptStackTrace(\"[CSheduler::ProcessStep2] stack trace:\\n\"))" "catch (...)" CODE "${CODE}")
    if(CODE MATCHES "__except")
        message(FATAL_ERROR "Unrecognized SEH syntax: update the fixture adapter explicitly")
    endif()
    file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/${VARIANT}")
    file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/${VARIANT}/scheduler-under-test.h" "${HEADER}")
    file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/${VARIANT}/scheduler-under-test.cpp" "${CODE}")
    add_executable(scheduler-${VARIANT} SchedulerTest.cpp)
    target_include_directories(scheduler-${VARIANT} PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/${VARIANT}")
    target_compile_features(scheduler-${VARIANT} PRIVATE cxx_std_17)
endforeach()
enable_testing()
add_test(NAME queue-equivalence COMMAND "${CMAKE_COMMAND}" "-DLEGACY=$<TARGET_FILE:scheduler-legacy>" "-DCURRENT=$<TARGET_FILE:scheduler-current>" -P "${CMAKE_CURRENT_SOURCE_DIR}/Compare.cmake")
