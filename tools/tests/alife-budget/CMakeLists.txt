cmake_minimum_required(VERSION 3.19)
project(alife-budget-test LANGUAGES CXX)
enable_testing()
set(source "${CMAKE_CURRENT_SOURCE_DIR}/../../../ogsr_engine/xrGame/alife_update_manager.cpp")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${source}")
file(READ "${source}" production)
string(REPLACE "\r\n" "\n" production "${production}")
string(FIND "${production}" "void CALifeUpdateManager::set_process_time(" start)
if(start EQUAL -1)
    message(FATAL_ERROR "Budget setter missing")
endif()
string(SUBSTRING "${production}" ${start} -1 tail)
string(FIND "${tail}" "\n}" end)
if(end EQUAL -1)
    message(FATAL_ERROR "Budget setter boundary missing")
endif()
math(EXPR length "${end} + 2")
string(SUBSTRING "${tail}" 0 ${length} method)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/budget.inc" "${method}\n")
add_executable(alife-budget-test BudgetTest.cpp)
target_compile_features(alife-budget-test PRIVATE cxx_std_17)
target_include_directories(alife-budget-test PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
add_test(NAME budget-units COMMAND alife-budget-test)

set(iterator "${CMAKE_CURRENT_SOURCE_DIR}/../../../ogsr_engine/xrGame/safe_map_iterator_inline.h")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${iterator}")
file(READ "${iterator}" implementation)
string(REPLACE "\r\n" "\n" implementation "${implementation}")
string(FIND "${implementation}" "IC u32 CSSafeMapIterator::update(" signature)
if(signature EQUAL -1)
    message(FATAL_ERROR "Iterator update signature missing")
endif()
string(SUBSTRING "${implementation}" 0 ${signature} prefix)
string(FIND "${prefix}" "template <" start REVERSE)
if(start EQUAL -1)
    message(FATAL_ERROR "Iterator update missing")
endif()
string(SUBSTRING "${implementation}" ${start} -1 tail)
string(FIND "${tail}" "\n}" end)
if(end EQUAL -1)
    message(FATAL_ERROR "Iterator update boundary missing")
endif()
math(EXPR length "${end} + 2")
string(SUBSTRING "${tail}" 0 ${length} method)
string(REPLACE "CSSafeMapIterator" "SliceIterator" method "${method}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/slice.inc" "${method}")
string(FIND "${implementation}" "IC bool CSSafeMapIterator::time_over(" start)
if(start EQUAL -1)
    message(FATAL_ERROR "Iterator time_over missing")
endif()
string(SUBSTRING "${implementation}" ${start} -1 tail)
string(FIND "${tail}" "}" end)
if(end EQUAL -1)
    message(FATAL_ERROR "Iterator time_over boundary missing")
endif()
math(EXPR length "${end} + 1")
string(SUBSTRING "${tail}" 0 ${length} method)
string(REPLACE "CSSafeMapIterator" "SliceIterator" method "${method}")
file(APPEND "${CMAKE_CURRENT_BINARY_DIR}/slice.inc" "\n${method}\n")
add_executable(alife-slice-test SliceTest.cpp)
target_compile_features(alife-slice-test PRIVATE cxx_std_17)
target_include_directories(alife-slice-test PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
add_test(NAME budget-progress COMMAND alife-slice-test)
