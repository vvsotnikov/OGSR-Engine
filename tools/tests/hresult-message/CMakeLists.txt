cmake_minimum_required(VERSION 3.19)
project(hresult-message-test LANGUAGES CXX)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(source "${CMAKE_CURRENT_SOURCE_DIR}/../../../ogsr_engine/xrCore/xrDebugNew.cpp")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${source}" "${CMAKE_CURRENT_SOURCE_DIR}/extract.py")
execute_process(COMMAND "${Python3_EXECUTABLE}" -B "${CMAKE_CURRENT_SOURCE_DIR}/extract.py"
    "${source}" "${CMAKE_CURRENT_BINARY_DIR}/method.inc" COMMAND_ERROR_IS_FATAL ANY)
add_executable(hresult-message-test test.cpp)
target_compile_features(hresult-message-test PRIVATE cxx_std_17)
target_include_directories(hresult-message-test PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(hresult-message-test PRIVATE comsuppw)
enable_testing()
add_test(NAME owned-hresult-message COMMAND hresult-message-test)
