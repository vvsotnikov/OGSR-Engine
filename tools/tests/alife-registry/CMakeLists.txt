cmake_minimum_required(VERSION 3.16)
project(alife-tests LANGUAGES CXX)
enable_testing()
function(extract_method file signature output)
    set(source "${CMAKE_CURRENT_SOURCE_DIR}/../../../ogsr_engine/xrGame/${file}")
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${source}")
    file(READ "${source}" production)
    string(REPLACE "\r\n" "\n" production "${production}")
    string(FIND "${production}" "${signature}" start)
    if(start EQUAL -1)
        message(FATAL_ERROR "${signature} missing; inspect fixture extraction")
    endif()
    string(SUBSTRING "${production}" ${start} -1 tail)
    string(FIND "${tail}" "\n}\n" end)
    if(end EQUAL -1)
        message(FATAL_ERROR "${signature} boundary missing; inspect fixture extraction")
    endif()
    math(EXPR length "${end} + 3")
    string(SUBSTRING "${tail}" 0 ${length} method)
    file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/${output}" "${method}")
endfunction()
