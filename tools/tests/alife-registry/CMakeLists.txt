cmake_minimum_required(VERSION 3.16)
project(alife-tests LANGUAGES CXX)
enable_testing()
set(source "${CMAKE_CURRENT_SOURCE_DIR}/../../../ogsr_engine/xrGame/alife_object_registry.cpp")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${source}")
file(READ "${source}" production)
string(FIND "${production}" "CALifeObjectRegistry::~CALifeObjectRegistry()" start)
if(start EQUAL -1)
    message(FATAL_ERROR "Registry destructor missing; inspect fixture extraction")
endif()
string(SUBSTRING "${production}" ${start} -1 tail)
string(FIND "${tail}" "\n}\n" end)
if(end EQUAL -1)
    message(FATAL_ERROR "Registry destructor boundary missing; inspect fixture extraction")
endif()
math(EXPR length "${end} + 3")
string(SUBSTRING "${tail}" 0 ${length} method)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/registry-destructor.inc" "${method}")
add_executable(registry-teardown-test RegistryTeardownTest.cpp)
target_compile_features(registry-teardown-test PRIVATE cxx_std_17)
target_include_directories(registry-teardown-test PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
add_test(NAME registry-teardown-lifetime COMMAND registry-teardown-test)
