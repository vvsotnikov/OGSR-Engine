cmake_minimum_required(VERSION 3.19)
project(alife-tests LANGUAGES CXX)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
enable_testing()
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/../../../ogsr_engine/xrGame/alife_switch_manager.cpp" "${CMAKE_CURRENT_SOURCE_DIR}/../../../ogsr_engine/xrGame/alife_update_manager.cpp" "${CMAKE_CURRENT_SOURCE_DIR}/../../../ogsr_engine/xrGame/alife_switch_manager.h" "${CMAKE_CURRENT_SOURCE_DIR}/Generate.py")
execute_process(COMMAND "${Python3_EXECUTABLE}" -B "${CMAKE_CURRENT_SOURCE_DIR}/Generate.py" "${CMAKE_CURRENT_BINARY_DIR}/methods.inc" "${CMAKE_CURRENT_BINARY_DIR}/metrics-update.inc" "${CMAKE_CURRENT_BINARY_DIR}/log-methods.inc" "${CMAKE_CURRENT_BINARY_DIR}/log-constants.inc" COMMAND_ERROR_IS_FATAL ANY)
foreach(pair "reconciliation-test|ReconciliationTest.cpp" "metrics-interval-test|MetricsIntervalTest.cpp")
 string(REPLACE "|" ";" parts "${pair}")
 list(GET parts 0 target)
 list(GET parts 1 source)
 add_executable(${target} ${source})
 target_compile_features(${target} PRIVATE cxx_std_17)
 target_include_directories(${target} PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
 add_test(NAME ${target} COMMAND ${target})
endforeach()

add_test(NAME reconciliation-log-reader COMMAND "${Python3_EXECUTABLE}" -B "${CMAKE_CURRENT_SOURCE_DIR}/../../profiling/Test-Reconciliation.py")

add_executable(reconciliation-log-test ReconciliationLogTest.cpp)
target_compile_features(reconciliation-log-test PRIVATE cxx_std_17)
target_include_directories(reconciliation-log-test PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
add_test(NAME reconciliation-log-format COMMAND reconciliation-log-test "${CMAKE_CURRENT_BINARY_DIR}/production.log")
set_tests_properties(reconciliation-log-format PROPERTIES FIXTURES_SETUP production_log)
add_test(NAME reconciliation-log-roundtrip COMMAND "${Python3_EXECUTABLE}" -B "${CMAKE_CURRENT_SOURCE_DIR}/CheckProductionLog.py" "${CMAKE_CURRENT_SOURCE_DIR}/../../profiling/Summarize-Reconciliation.py" "${CMAKE_CURRENT_BINARY_DIR}/production.log")
set_tests_properties(reconciliation-log-roundtrip PROPERTIES FIXTURES_REQUIRED production_log)
