cmake_minimum_required(VERSION 3.19)
project(alife-tests LANGUAGES CXX)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
enable_testing()
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/../../../ogsr_engine/xrGame/alife_switch_manager.cpp" "${CMAKE_CURRENT_SOURCE_DIR}/../../../ogsr_engine/xrGame/alife_update_manager.cpp" "${CMAKE_CURRENT_SOURCE_DIR}/Generate.py" "${CMAKE_CURRENT_SOURCE_DIR}/Baseline.inc")
execute_process(COMMAND "${Python3_EXECUTABLE}" -B "${CMAKE_CURRENT_SOURCE_DIR}/Generate.py" "${CMAKE_CURRENT_BINARY_DIR}/methods.inc" "${CMAKE_CURRENT_BINARY_DIR}/metrics-update.inc" COMMAND_ERROR_IS_FATAL ANY)
foreach(pair "reconciliation-test|ReconciliationTest.cpp" "metrics-interval-test|MetricsIntervalTest.cpp")
 string(REPLACE "|" ";" parts "${pair}")
 list(GET parts 0 target)
 list(GET parts 1 source)
 add_executable(${target} ${source})
 target_compile_features(${target} PRIVATE cxx_std_17)
 target_include_directories(${target} PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
 add_test(NAME ${target} COMMAND ${target})
endforeach()
