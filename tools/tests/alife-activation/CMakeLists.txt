cmake_minimum_required(VERSION 3.16)
project(alife-activation-tests LANGUAGES CXX)
enable_testing()
add_executable(activation-test ActivationTest.cpp)
target_compile_features(activation-test PRIVATE cxx_std_17)
add_test(NAME activation-queue COMMAND activation-test)

set(source "${CMAKE_CURRENT_SOURCE_DIR}/../../../ogsr_engine/xrGame/alife_switch_manager.cpp")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${source}")
file(READ "${source}" production)
string(REPLACE "\r\n" "\n" production "${production}")
set(methods "")
foreach(name try_switch_online request_switch_online begin_activation_collection finish_activation_collection switch_object)
    string(FIND "${production}" "void CALifeSwitchManager::${name}(" start)
    if(start EQUAL -1)
        message(FATAL_ERROR "Activation manager method missing: ${name}")
    endif()
    string(SUBSTRING "${production}" ${start} -1 tail)
    string(FIND "${tail}" "\n}" end)
    if(end EQUAL -1)
        message(FATAL_ERROR "Activation manager method boundary missing: ${name}")
    endif()
    math(EXPR length "${end} + 2")
    string(SUBSTRING "${tail}" 0 ${length} method)
    string(APPEND methods "${method}\n")
endforeach()
string(REPLACE "CALifeSwitchManager::" "Manager::" methods "${methods}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/online-method.inc" "${methods}")
set(header "${CMAKE_CURRENT_SOURCE_DIR}/../../../ogsr_engine/xrGame/alife_switch_manager.h")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${header}")
file(READ "${header}" declarations)
string(REGEX MATCH "static constexpr unsigned activation_attempt_limit = [0-9]+;" limit "${declarations}")
if(NOT limit)
    message(FATAL_ERROR "Activation attempt limit missing")
endif()
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/limit.inc" "${limit}")
add_executable(activation-integration-test IntegrationTest.cpp)
target_compile_features(activation-integration-test PRIVATE cxx_std_17)
target_include_directories(activation-integration-test PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
add_test(NAME activation-integration COMMAND activation-integration-test)
