cmake_minimum_required(VERSION 3.16)
project(alife-activation-tests LANGUAGES CXX)
enable_testing()
add_executable(activation-test ActivationTest.cpp)
target_compile_features(activation-test PRIVATE cxx_std_17)
add_test(NAME activation-queue COMMAND activation-test)

set(source "${CMAKE_CURRENT_SOURCE_DIR}/../../../ogsr_engine/xrGame/alife_switch_manager.cpp")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${source}")
file(READ "${source}" production)
string(REPLACE "\r\n" "\n" production "${production}")
string(FIND "${production}" "void CALifeSwitchManager::try_switch_online(" start)
if(start EQUAL -1)
    message(FATAL_ERROR "Activation manager method missing")
endif()
string(SUBSTRING "${production}" ${start} -1 tail)
string(FIND "${tail}" "\n}" end)
if(end EQUAL -1)
    message(FATAL_ERROR "Activation manager method boundary missing")
endif()
math(EXPR length "${end} + 2")
string(SUBSTRING "${tail}" 0 ${length} method)
string(REPLACE "CALifeSwitchManager::" "Manager::" method "${method}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/online-method.inc" "${method}")
add_executable(activation-integration-test IntegrationTest.cpp)
target_compile_features(activation-integration-test PRIVATE cxx_std_17)
target_include_directories(activation-integration-test PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
add_test(NAME activation-client-state COMMAND activation-integration-test)
