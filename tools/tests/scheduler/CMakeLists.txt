cmake_minimum_required(VERSION 3.16)
project(scheduler-behavior LANGUAGES CXX)
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
    "${ROOT}/ogsr_engine/xr_3da/xrSheduler.h" "${ROOT}/ogsr_engine/xr_3da/xrSheduler.cpp")
file(READ "${ROOT}/ogsr_engine/xr_3da/xrSheduler.h" HEADER)
string(REPLACE "#include \"ISheduled.h\"" "" HEADER "${HEADER}")
file(READ "${ROOT}/ogsr_engine/xr_3da/xrSheduler.cpp" CODE)
string(REGEX REPLACE "#include[^\n]*" "" CODE "${CODE}")
# Model the exact native finally block with C++ cleanup on normal/exceptional exit.
string(REPLACE "__finally\n    {\n        FinishStep();\n    }"
    "catch (...) { FinishStep(); throw; }\n    FinishStep();" CODE "${CODE}")
if(CODE MATCHES "__finally")
    message(FATAL_ERROR "Unrecognized finally syntax: update the fixture adapter explicitly")
endif()
string(REPLACE "__try" "try" CODE "${CODE}")
string(REPLACE "__except (ExceptStackTrace(\"[CSheduler::ProcessStep] stack trace:\\n\"))" "catch (...)" CODE "${CODE}")
string(REPLACE "__except (ExceptStackTrace(\"[CSheduler::ProcessStep2] stack trace:\\n\"))" "catch (...)" CODE "${CODE}")
if(CODE MATCHES "__except")
    message(FATAL_ERROR "Unrecognized SEH syntax: update the fixture adapter explicitly")
endif()
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/scheduler-under-test.h" "${HEADER}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/scheduler-under-test.cpp" "${CODE}")
enable_testing()
set(CASES normal_cancels_rt scale_lifecycle step_cleanup exception_cleanup rt_mutation ordering compaction exceptions liveness intervals budget prefetch needed self_remove remove_other registration realtime stress)
foreach(VARIANT release debug)
    set(TARGET scheduler-${VARIANT}-test)
    add_executable(${TARGET} SchedulerTest.cpp)
    target_include_directories(${TARGET} PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
    target_compile_features(${TARGET} PRIVATE cxx_std_17)
    if(VARIANT STREQUAL "debug")
        target_compile_definitions(${TARGET} PRIVATE DEBUG)
    endif()
    foreach(CASE IN LISTS CASES)
        add_test(NAME scheduler-${VARIANT}-${CASE} COMMAND ${TARGET} ${CASE})
    endforeach()
endforeach()
add_test(NAME scheduler-debug-registered_state COMMAND scheduler-debug-test registered_state)

add_test(NAME scheduler-debug-registered_consistency COMMAND scheduler-debug-test registered_consistency)
