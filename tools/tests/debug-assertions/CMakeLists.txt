cmake_minimum_required(VERSION 3.19)
project(debug-assertions LANGUAGES CXX)
enable_testing()
foreach(variant enabled required suppressed release)
    add_executable(assertions-${variant} AssertionTest.cpp)
    if(NOT variant STREQUAL "release")
        target_compile_definitions(assertions-${variant} PRIVATE ENABLE_DEBUG_CASE)
    endif()
    if(variant STREQUAL "required")
        target_compile_definitions(assertions-${variant} PRIVATE OGSR_REQUIRE_DEBUG_ASSERTIONS)
    elseif(variant STREQUAL "suppressed")
        target_compile_definitions(assertions-${variant} PRIVATE DISABLE_DBG_ASSERTIONS)
    endif()
    add_test(NAME assertions-${variant} COMMAND assertions-${variant})
endforeach()

# Required validation must reject both suppression and a non-Debug compiler mode.
foreach(variant suppressed release)
    set(definitions -DOGSR_REQUIRE_DEBUG_ASSERTIONS)
    if(variant STREQUAL "suppressed")
        list(APPEND definitions -DENABLE_DEBUG_CASE -DDISABLE_DBG_ASSERTIONS)
    endif()
    try_compile(compiled "${CMAKE_CURRENT_BINARY_DIR}/reject-${variant}"
        SOURCES "${CMAKE_CURRENT_SOURCE_DIR}/AssertionTest.cpp"
        COMPILE_DEFINITIONS ${definitions} OUTPUT_VARIABLE output)
    if(compiled OR NOT output MATCHES "Local validation requires Debug assertions")
        message(FATAL_ERROR "Required ${variant} assertions were not rejected correctly: ${output}")
    endif()
endforeach()
